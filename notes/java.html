Practical 2: Classes, Objects and Constructors

Create a Student class with: Data members, Constructors (default and parameterized) and Methods to display details.

class Student {
    int rollNo;
    String name;
    // Default Constructor
    Student() {
        rollNo = 0;
        name = "Unknown";
    }
    // Parameterized Constructor
    Student(int r, String n) {
        rollNo = r;
        name = n;
    }
    void display() {
        System.out.println("Roll No: " + rollNo);
        System.out.println("Name: " + name);
    }
    public static void main(String[] args) {
        Student s1 = new Student();
        Student s2 = new Student(101, "Swara");
        s1.display();
        s2.display();
    }
}

/* Output:

Name: Unknown
Roll No: 101
Name: Swara

Practical 5: Polymorphism

Implement: Method Overloading and Method Overriding.

// A) Method Overloading
class Addition {
    int add(int a, int b) {
        return a + b;
    }
    int add(int a, int b, int c) {
        return a + b + c;
    }
    public static void main(String[] args) {
        Addition obj = new Addition();
        System.out.println(obj.add(10, 20));
        System.out.println(obj.add(10, 20, 30));
    }
}

// B) Method Overriding
class Animal {
    void sound() {
        System.out.println("Animal makes sound");
    }
}
class Dog extends Animal {
    void sound() {
        System.out.println("Dog barks");
    }
    public static void main(String[] args) {
        Dog d = new Dog();
        d.sound();
    }
}

/* Output:
A) 
30
60

B) 
Dog barks
*/

Create Student Information Form using: JFrame, JPanel, JLabel, JTextField and JButton.

import javax.swing.*;
import java.awt.event.*;
public class StudentForm {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Student Information Form");
        JPanel panel = new JPanel();
        JLabel nameLabel = new JLabel("Name:");
        JTextField nameField = new JTextField(15);
        JLabel courseLabel = new JLabel("Course:");
        JTextField courseField = new JTextField(15);
        JButton button = new JButton("Submit");
        
        panel.add(nameLabel);
        panel.add(nameField);
        panel.add(courseLabel);
        panel.add(courseField);
        panel.add(button);
        frame.add(panel);
        
        button.addActionListener(new ActionListener() {
            public void actionPerformed(ActionEvent e) {
                JOptionPane.showMessageDialog(frame,
                "Name: " + nameField.getText() +
                "\nCourse: " + courseField.getText());
            }
        });
        
        frame.setSize(300, 200);
        frame.setVisible(true);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
    }
}

Practical 11: JDBC CRUD Application

Connect Java with MySQL/SQL Server and perform: Insert, Update, Delete and Select using PreparedStatement.

/* SQL Table
CREATE TABLE student(
    id INT PRIMARY KEY,
    name VARCHAR(50),
    course VARCHAR(50)
);
*/

import java.sql.*;
public class JDBCDemo {
    public static void main(String[] args) {
        try {
            Class.forName("com.mysql.cj.jdbc.Driver");
            Connection con = DriverManager.getConnection(
            "jdbc:mysql://localhost:3306/test", "root", "root");
            
            // INSERT
            PreparedStatement ps1 = con.prepareStatement("insert into student values(?,?,?)");
            ps1.setInt(1, 1);
            ps1.setString(2, "Amit");
            ps1.setString(3, "BCA");
            ps1.executeUpdate();
            System.out.println("Record Inserted");
            
            // UPDATE
            PreparedStatement ps2 = con.prepareStatement("update student set course=? where id=?");
            ps2.setString(1, "MCA");
            ps2.setInt(2, 1);
            ps2.executeUpdate();
            System.out.println("Record Updated");

            // DELETE
            PreparedStatement ps3 = con.prepareStatement("delete from student where id=?");
            ps3.setInt(1, 1);
            ps3.executeUpdate();
            System.out.println("Record Deleted");
            
            // SELECT
            Statement st = con.createStatement();
            ResultSet rs = st.executeQuery("select * from student");
            while(rs.next()) {
                System.out.println(rs.getInt(1) + " " + rs.getString(2) + " " + rs.getString(3));
            }
            con.close();
        } catch(Exception e) {
            System.out.println(e);
        }
    }
}

/* Output:
Record Inserted
Record Updated
Record Deleted
*/

Practical 13: JSON with Java

Create and parse JSON data using Java.

import org.json.simple.JSONObject;
import org.json.simple.parser.JSONParser;
public class JSONDemo {
    public static void main(String[] args) {
        try {
            // Create JSON
            JSONObject obj = new JSONObject();
            obj.put("id", 101);
            obj.put("name", "Amit");
            obj.put("course", "BCA");
            String jsonData = obj.toJSONString();
            System.out.println("Created JSON:");
            System.out.println(jsonData);
            
            // Parse JSON
            JSONParser parser = new JSONParser();
            JSONObject data = (JSONObject) parser.parse(jsonData);
            System.out.println("\nParsed JSON Data:");
            System.out.println("ID: " + data.get("id"));
            System.out.println("Name: " + data.get("name"));
            System.out.println("Course: " + data.get("course"));
        } catch(Exception e) {
            System.out.println(e);
        }
    }
}

/* Output:
Created JSON:
{"id":101,"name":"Amit","course":"BCA"}

Parsed JSON Data:
ID: 101
Name: Amit
Course: BCA
*/
